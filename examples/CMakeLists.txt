# Example programs

add_executable(simple_note simple_note.c)
target_include_directories(simple_note PRIVATE ${CMAKE_SOURCE_DIR}/include)
if(TARGET midix_static)
    target_link_libraries(simple_note PRIVATE midix_static)
else()
    target_link_libraries(simple_note PRIVATE midix_shared)
endif()
